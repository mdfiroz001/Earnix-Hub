                        tg.showAlert("Failed to submit request. Your balance has been restored.");
                    }
                } finally {

                    btn.disabled = false;

                    return; 
                }
                
                data.forEach((user, index) => {
                    const score = activeBoard === 'referral' ? (user.referrals || 0) : (user.totalEarned || 0).toFixed(2);
                    const item = document.createElement('div'); 
                    item.className = 'rank-card';
                    item.innerHTML = `
                        <span class="rank-number">#${index + 1}</span>
                        <img class="rank-avatar" src="${user.photoUrl || 'https://via.placeholder.com/55'}" alt="User">
                        <div class="rank-info">
                            <div class="rank-name">${user.firstName || 'User'}</div>
                            <div class="rank-score">${score}</div>
                        </div>
                    `;
                    listEl.appendChild(item);
                });
            };

            const renderDynamicTasks = async () => {
                const container = document.getElementById('dynamic-tasks-container'); 
                if (!container) return;
                
                container.innerHTML = '';
                
                const result = await db.execute("SELECT * FROM bonus_tasks");
                const tasks = result.data;
                tasks.reverse();
                
                tasks.forEach(task => {
                    const isCompleted = userState.completedTasks && userState.completedTasks[task.id];
                    const el = document.createElement('div'); 
                    el.className = 'bonus-task-card';
                    el.innerHTML = `
                        <div class="bonus-icon"><i class="fas fa-gift"></i></div>
                        <div class="bonus-info">
                            <h4>${task.name}</h4>
                            <p>Get ${task.reward} TK</p>
                        </div>
                        <button class="bonus-btn" data-task-id="${task.id}" data-task-url="${task.url}" ${isCompleted ? 'disabled' : ''}>
                            ${isCompleted ? 'Claimed' : 'Claim'}
                        </button>
                    `;
                    container.appendChild(el);
                    
                    const btn = el.querySelector('.bonus-btn');
                    if (btn && !isCompleted) {
                        btn.addEventListener('click', handleClaimTask);
                    }
                });
            };

            const renderReferralSection = () => {
                const container = document.getElementById('referral-section');
                if (!container) return;
                
                const botUsername = appConfig.botUsername || 'app';
                const refLink = `https://t.me/${botUsername}/app?startapp=${tgUser.id}`;
                
                container.innerHTML = `
                    <div class="referral-box">
                        <i class="fas fa-user-plus" style="font-size: 3rem; color: var(--primary-500); margin-bottom: 15px;"></i>
                        <h3 style="margin-bottom: 10px;">Invite Friends</h3>
                        <p style="margin: 10px 0;">Earn ${appConfig.referralBonus || 0} TK per referral</p>
                        <div class="referral-link" id="referral-link">${refLink}</div>
                        <button id="copy-ref-link-btn" class="bonus-btn" style="width: 100%; padding: 16px;">Copy Link</button>
                    </div>
                `;
                
                document.getElementById('copy-ref-link-btn')?.addEventListener('click', copyReferralLink);
            };

            const renderWithdrawSection = () => {
                const container = document.getElementById('withdraw-section');
                if (!container) return;
                
                let optionsHtml = '';
                if (appConfig.withdrawMethods) { 
                    appConfig.withdrawMethods.forEach(method => { 
                        optionsHtml += `<option value="${method.name}">${method.name} (Min: ${method.min} TK)</option>`; 
                    }); 
                }

                const minRefs = appConfig.minimumWithdrawReferrals || 0;
                let referralNotice = '';
                if (minRefs > 0) {
                    referralNotice = `
                        <p style="font-size: 0.9rem; margin-bottom: 20px; padding: 12px; background: rgba(245, 158, 11, 0.1); border-radius: 20px; color: #F59E0B;">
                            <i class="fas fa-info-circle" style="margin-right: 8px;"></i> Need at least ${minRefs} referrals to withdraw
                        </p>
                    `;
                }

                container.innerHTML = `
                    <form id="withdraw-form" class="glass-form">
                        <h3 style="margin-bottom: 25px; text-align: center;">Withdraw Funds</h3>
                        ${referralNotice}
                        <select id="withdraw-method" class="form-input" required>
                            ${optionsHtml}
                        </select>
                        <input type="text" id="account-number" class="form-input" placeholder="Account Number / Wallet ID" required>
                        <input type="number" step="0.01" id="amount" class="form-input" placeholder="Amount (TK)" required>
                        <button type="submit" id="withdraw-submit-btn" class="bonus-btn" style="width: 100%; padding: 16px;">Submit Request</button>
                    </form>
                `;
                
                document.getElementById('withdraw-form')?.addEventListener('submit', handleWithdraw);
            };

            const renderDynamicLinks = async () => {
                const container = document.getElementById('dynamic-links-container'); 
                if (!container) return;
                
                container.innerHTML = '';
                
                const result = await db.execute("SELECT * FROM links");
                const links = result.data;
                links.reverse();
                
                links.forEach(link => {
                    const el = document.createElement('div'); 
                    el.className = 'link-card';
                    el.innerHTML = `
                        <div class="link-icon"><i class="fas fa-link"></i></div>
                        <div style="flex: 1;">
                            <h4 style="font-weight: 700;">${link.name}</h4>
                        </div>
                        <button class="bonus-btn" style="padding: 10px 20px;">Visit</button>
                    `;
                    
                    const btn = el.querySelector('.bonus-btn');
                    btn.addEventListener('click', (e) => {
                        e.stopPropagation();
                        if (tg) tg.openLink(link.url);
                    });
                    
                    container.appendChild(el);
                });
            };

            initApp();
        });
